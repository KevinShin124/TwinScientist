    Called after evolution_manager when the research pipeline has
    finished generating its report. This node enables open-ended
    scientific discussion — the kind that happens in real labs
    long after the initial analysis phase.

    Flow:
    1. Extract accumulated state summary (report, hypotheses, evidence)
    2. Load last N chat turns from history for continuity
    3. Detect if there's an interrupt message via resume_value injection
    4. Call LLM with combined context + user intent
    5. Generate structured response (reply text + detected action)
    6. Route based on detected intent:
       - challenge/re_analyze → loop_back_to_X
       - continue/end → return final signal
       - question/explore → reply & stay in this node

def _detect_intent(message: str | None) -> str:
    """Detect what the user intends based on their message."""
    if not message:
        return "unknown"

    msg_lower = message.lower()

    # Re-analysis requests → route back to analysis pipelines
    reanalyze_signals = ["重新分析", "re-analyse", "re-analyze", "reanalyze", "换个角度",
                         "从不同角度", "another angle", "换一种方法", "change method",
                         "试一下另一种方式", "再分析一遍", "use different approach"]
    if any(s in msg_lower for s in reanalyze_signals):
        return "request_re_analysis"

    # New hypothesis generation
    new_hyp_signals = ["新的假设", "new hypothesis", "generate new", "提出新假说",
                       "change direction", "调整方向", "换个思路"]
    if any(s in msg_lower for s in new_hyp_signals):
        return "request_new_hypothesis"

    # Challenge/critique
    challenge_signals = ["质疑", "challenge", "wrong", "错误", "质疑你的结论",
                         "有问题", "我不同意", "不对", "flawed", "漏洞"]
    if any(s in msg_lower for s in challenge_signals):
        return "challenge_findings"

    # Deepen understanding
    deepen_signals = ["解释一下", "explain", "详细说明", "深入一点", "tell me more",
                      "为什么", "why did you", "是什么意思", "clarify"]
    if any(s in msg_lower for s in deepen_signals):
        return "deepen_understanding"

    # End/wrap up
    end_signals = ["结束", "end", "ok", "可以了", "够了", "accepted", "wrap up",
                   "done", "可以了就行", "就这样吧"]
    if any(s in msg_lower for s in end_signals):
        return "accept_and_end"

    return "general_question"
